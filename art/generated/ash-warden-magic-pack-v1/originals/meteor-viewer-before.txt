import {sampleStrike,sampleCast,strikesFor} from './timing.mjs';
const {document,Image}=globalThis,meta=globalThis.METEOR_CANDIDATES,canvas=document.querySelector('canvas'),ctx=canvas.getContext('2d');
const ui=Object.fromEntries(['mode','play','restart','sample','rate','zoom','time','loop','shield','status'].map(id=>[id,document.getElementById(id)]));
const images={},scaled=new Map(),total=3100;let time=0,last=0,playing=true,raf,ground,ready=false,completed=false;
const idle='../ash-warden-directional-melee-b-v1/south/idle/frames/00.png';
function panels(){return ui.mode.value==='casts'?[{label:'Cast A · Compact Summon',cast:'a'},{label:'Cast B · Strong Summon',cast:'b'}]:ui.mode.value==='pairings'?[{label:'Cast A + Meteor A',cast:'a',meteor:'a'},{label:'Cast A + Meteor B',cast:'a',meteor:'b'},{label:'Cast B + Meteor C',cast:'b',meteor:'c'}]:['a','b','c'].map(k=>({label:'Meteor '+k.toUpperCase()+' · '+meta.meteors[k].name,meteor:k}));}
function stamp(url,cx,cy,pivot,zoom){const key=url+'@'+zoom;if(!scaled.has(key)){const im=images[url],tile=document.createElement('canvas');tile.width=Math.round(im.width*meta.worldScale*zoom);tile.height=Math.round(im.height*meta.worldScale*zoom);const x=tile.getContext('2d');x.imageSmoothingEnabled=false;x.drawImage(im,0,0,tile.width,tile.height);scaled.set(key,tile);}const s=meta.worldScale*zoom;ctx.drawImage(scaled.get(key),Math.round(cx-pivot[0]*s),Math.round(cy-pivot[1]*s));}
function drawComponent(meteor,part,sample,x,y,zoom){const index=sample[part];if(index<0)return;const clip=meteor.components[part],offset=part==='fall'?sample.fallOffset:[0,0];stamp(clip.frames[index],x+offset[0]*zoom,y+offset[1]*zoom,clip.pivot,zoom);}
function draw(){if(!ready)return;const p=panels(),w=canvas.width,h=canvas.height,pw=w/p.length,zoom=Number(ui.zoom.value);ctx.imageSmoothingEnabled=false;
 p.forEach((panel,i)=>{const x=i*pw,cy=h*.68,cx=x+pw/2;ctx.save();ctx.beginPath();ctx.rect(x,0,pw,h);ctx.clip();const tw=ground.width*.5*zoom,th=ground.height*.5*zoom;ctx.fillStyle='#343b2d';ctx.fillRect(x,0,pw,h);for(let yy=0;yy<h;yy+=th)for(let xx=x;xx<x+pw;xx+=tw)ctx.drawImage(ground,Math.round(xx),Math.round(yy),Math.ceil(tw),Math.ceil(th));
  const tx=panel.cast&&panel.meteor?x+pw*.72:cx,bx=panel.meteor?x+pw*.25:cx,by=cy;
  const meteor=meta.meteors[panel.meteor],strikes=panel.meteor?strikesFor(ui.mode.value):[];
  for(const strike of strikes){const sample=sampleStrike(meteor,time,strike.spawnMs);for(const part of ['telegraph','aftermath'])drawComponent(meteor,part,sample,tx+strike.offset[0]*zoom,cy+strike.offset[1]*zoom,zoom);}
  if(panel.cast){const cast=meta.casts[panel.cast],index=sampleCast(cast,time);if(ui.shield.checked){ctx.strokeStyle='#a6bbb5';ctx.lineWidth=1;ctx.beginPath();ctx.ellipse(bx,by-38*zoom,43*zoom,49*zoom,0,0,Math.PI*2);ctx.stroke();}stamp(index<0?idle:cast.frames[index],bx,by,[96,154],zoom);}
  for(const strike of strikes){const sample=sampleStrike(meteor,time,strike.spawnMs);for(const part of ['fall','impact'])drawComponent(meteor,part,sample,tx+strike.offset[0]*zoom,cy+strike.offset[1]*zoom,zoom);}
  ctx.fillStyle='#14252cf0';ctx.fillRect(x+4,4,pw-8,38);ctx.fillStyle='#fff0d2';ctx.textAlign='center';ctx.font=(w<900?'10':'12')+'px Consolas,monospace';const label=panel.label.split(' · ');ctx.fillText(label[0],cx,19);ctx.fillText(label[1]??'Независимые assets',cx,34);ctx.restore();ctx.strokeStyle='#687363';ctx.strokeRect(x+.5,.5,pw-1,h-1);
 });ui.time.value=Math.round(time);ui.status.textContent=Math.round(time)+'ms';}
function resize(){canvas.width=globalThis.innerWidth;canvas.height=Math.max(100,globalThis.innerHeight-document.querySelector('header').offsetHeight-document.querySelector('p').offsetHeight);draw();}
function seek(t){time=Math.max(0,Math.min(total,t));last=0;completed=false;draw();}
function tick(t){const dt=last?Math.min(100,t-last):0;last=t;if(playing){time=Math.min(total,time+dt*Number(ui.rate.value));if(time===total){completed=true;if(ui.loop.checked)time=0;else playing=false;}}draw();raf=globalThis.requestAnimationFrame(tick);}
ui.play.onclick=()=>{playing=!playing;last=0;};ui.restart.onclick=()=>{seek(0);playing=true;};ui.mode.onchange=()=>{draw();};ui.zoom.onchange=resize;ui.shield.onchange=draw;ui.rate.onchange=()=>{last=0;};ui.time.oninput=()=>{playing=false;seek(Number(ui.time.value));};ui.sample.onchange=()=>{playing=false;seek(Number(ui.sample.value));};
globalThis.addEventListener('resize',resize);globalThis.addEventListener('pagehide',()=>globalThis.cancelAnimationFrame(raf));document.addEventListener('visibilitychange',()=>{last=0;});
globalThis.meteorReviewState=()=>({ready,time,playing,completed,mode:ui.mode.value,panels:panels(),casts:Object.fromEntries(Object.entries(meta.casts).map(([k,c])=>[k,sampleCast(c,time)])),strikes:Object.fromEntries(Object.entries(meta.meteors).map(([k,m])=>[k,strikesFor(ui.mode.value).map(s=>sampleStrike(m,time,s.spawnMs))])),zoom:Number(ui.zoom.value),selection:meta.selection});
(async()=>{const urls=new Set([idle,...Object.values(meta.casts).flatMap(c=>c.frames),...Object.values(meta.meteors).flatMap(m=>Object.values(m.components).flatMap(c=>c.frames))]);for(const url of urls){const im=new Image();im.src=url;await im.decode();images[url]=im;}ground=new Image();ground.src=globalThis.LOCOMOTION_REVIEW.ground;await ground.decode();ready=true;resize();raf=globalThis.requestAnimationFrame(tick);})().catch(e=>{ui.status.textContent=e.message;throw e;});
